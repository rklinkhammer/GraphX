# Phase 2A: Configuration Management Tests (Catch2)
# These tests verify FHSS configuration derivation, validation, and state management
# Always included (not dependent on web dashboard)

find_package(Catch2 REQUIRED)
find_package(nlohmann_json REQUIRED)

# Phase 2A: Configuration Management Library
# Collects implementation files for FHSSConfigurationDeriver, FHSSCrossNodeValidator, ConfigurationStateMachine
file(GLOB PHASE2A_IMPL_SOURCES CONFIGURE_DEPENDS 
    "${CMAKE_CURRENT_SOURCE_DIR}/../src/configuration/*.cpp"
)

if(NOT PHASE2A_IMPL_SOURCES)
    # Create empty library as placeholder if no implementation yet
    add_library(libdsp_configuration INTERFACE)
    target_include_directories(libdsp_configuration INTERFACE
        ${CMAKE_CURRENT_SOURCE_DIR}/../include
    )
    target_link_libraries(libdsp_configuration INTERFACE dsp nlohmann_json::nlohmann_json)
else()
    add_library(libdsp_configuration STATIC ${PHASE2A_IMPL_SOURCES})
    target_include_directories(libdsp_configuration PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/../include
    )
    target_link_libraries(libdsp_configuration PUBLIC
        dsp
        graph
        nlohmann_json::nlohmann_json
    )
    target_link_libraries(libdsp_configuration PRIVATE graphx_project_options)
    if(UNIX AND NOT APPLE)
        find_library(GRAPHX_UUID_LIBRARY NAMES uuid REQUIRED)
        target_link_libraries(libdsp_configuration PUBLIC ${GRAPHX_UUID_LIBRARY})
    endif()
    target_compile_options(libdsp_configuration PRIVATE
        $<$<CXX_COMPILER_ID:MSVC>:/W4>
        $<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-Wall -Wextra -Wpedantic>
    )
endif()

# Collect Phase 2A test files (unit directory for FHSSConfigurationDeriver, FHSSCrossNodeValidator, ConfigurationStateMachine)
set(PHASE2A_TEST_SOURCES
    "unit/test_fhss_configuration_deriver.cpp"
    "unit/test_fhss_cross_node_validator.cpp"
    "unit/test_fhss_configuration_state_machine.cpp"
    "integration/test_fhss_http_cli_integration.cpp"
)

if(PHASE2A_TEST_SOURCES)
    add_executable(test_phase2a_configuration ${PHASE2A_TEST_SOURCES})
    
    target_include_directories(test_phase2a_configuration PRIVATE
        ${CMAKE_SOURCE_DIR}/libgraph/test/include
        ${CMAKE_SOURCE_DIR}/libdsp/include
        ${CMAKE_CURRENT_SOURCE_DIR}/fixtures
    )
    
    target_compile_definitions(test_phase2a_configuration PRIVATE
        GRAPHX_SOURCE_ROOT="${CMAKE_SOURCE_DIR}"
        PLUGIN_OUTPUT_DIRECTORY="${CMAKE_BINARY_DIR}/plugins"
        PHASE2A_FIXTURES_PATH="${CMAKE_CURRENT_SOURCE_DIR}/fixtures"
    )
    
    target_link_libraries(test_phase2a_configuration PRIVATE
        libdsp_configuration
        dsp
        graph
        gpu
        Catch2::Catch2WithMain
        nlohmann_json::nlohmann_json
        Threads::Threads
        graphx_project_options
    )
    
    # Register with CTest
    add_test(NAME phase2a_configuration COMMAND test_phase2a_configuration)
    set_tests_properties(phase2a_configuration PROPERTIES 
        LABELS "phase2a;configuration;catch2;ci-safe;host-any"
    )
    
    # Catch2 discovery test (optional - verify test discovery works)
    add_test(NAME phase2a_configuration_discover
        COMMAND ${CMAKE_COMMAND}
            -DTEST_EXECUTABLE=$<TARGET_FILE:test_phase2a_configuration>
            -DCATCH2_TEST=1
            -P ${CMAKE_SOURCE_DIR}/cmake/CheckGTestDiscovery.cmake
    )
    set_tests_properties(phase2a_configuration_discover PROPERTIES
        LABELS "phase2a;configuration;discovery;catch2;ci-safe;host-any"
        DISABLED TRUE  # Disabled since GTest discovery doesn't apply to Catch2
    )
    
    # Add custom target for running phase2a tests
    add_custom_target(test_phase2a
        COMMAND ${CMAKE_CTEST_COMMAND} -R "^phase2a_configuration$" --output-on-failure
        COMMENT "Running Phase 2A configuration tests"
    )
    
    message(STATUS "Phase 2A Catch2 tests enabled: ${PHASE2A_TEST_SOURCES}")
else()
    message(WARNING "No Phase 2A test sources found in integration/")
endif()
